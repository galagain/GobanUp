cmake_minimum_required(VERSION 3.13)
project(FuegoBrowser CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fexceptions -Wno-deprecated-declarations -Wno-c++11-narrowing")
include_directories(/boost)
foreach(lib gtpengine smartgame go gouct)
  include_directories(../${lib})
  add_subdirectory(../${lib} ${lib})
endforeach()
add_executable(engine fuego.cpp)
target_link_libraries(engine fuego_gouct)
target_link_options(engine PRIVATE --no-entry -fexceptions -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=8388608 -sMODULARIZE=1 -sEXPORT_NAME=FuegoModule -sENVIRONMENT=worker "-sEXPORTED_FUNCTIONS=['_choose','_malloc','_free']" "-sEXPORTED_RUNTIME_METHODS=['HEAP32']")
